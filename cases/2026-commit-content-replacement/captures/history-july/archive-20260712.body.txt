      items[i] = items[j];
      items[j] = temp;
    }

/* ===============================

const SLOT_PROVIDERS = [
  { provider:"Pragmatic Play", code:"PP", jam:"00:00 - 06:00", status:"HOT" },
  { provider:"PG Soft", code:"PG", jam:"13:00 - 18:00", status:"STABLE" },
  { provider:"Habanero", code:"HB", jam:"19:00 - 23:00", status:"BOOST" },
  { provider:"Joker Gaming", code:"JK", jam:"10:00 - 14:00", status:"TRENDING" },
  { provider:"Spadegaming", code:"SG", jam:"15:00 - 20:00", status:"HOT" },
  { provider:"Microgaming", code:"MG", jam:"21:00 - 01:00", status:"UP" }
];

const SLOT_PATTERNS = [
  "AUTO SPIN 20x",
  "MANUAL 10x",
  "TURBO 30x",
  "BUY FEATURE",
  "SPIN CEPAT",
  "BET SELANG"
];

const SLOT_STATUS = [
  "GACOR",
  "PANAS",
  "STABIL",
  "NAIK",
  "RAME",
  "MAXWIN MODE"
];

const $ = (q) => document.querySelector(q);
const randInt = (min,max) => Math.floor(Math.random()*(max-min+1))+min;

function uniqueProviders(){
  return SLOT_PROVIDERS.map(p => p.provider);
}

function simulateRtp(){
  return randInt(80, 98);
}

function getRtpText(value){
  return value + "%";
}

function getSpinMeter(){
  return randInt(120, 980) + " SPIN";
}

function getRandomPattern(){
  return SLOT_PATTERNS[randInt(0, SLOT_PATTERNS.length - 1)];
}

function getRandomStatus(){
  return SLOT_STATUS[randInt(0, SLOT_STATUS.length - 1)];
}

function applyFilters(){
  const provider = $("#providerFilter").value;
  const minRtp = parseFloat($("#minRtp").value || "0");

  return SLOT_PROVIDERS
    .filter(x => (provider === "ALL" || x.provider === provider))
    .map(x => ({ ...x, rtp: simulateRtp() }))
    .filter(x => x.rtp >= minRtp);
}

function pickRandom(arr, n=1){
  const copy = [...arr];
  const out = [];
  for(let i=0;i<n && copy.length;i++){
    const idx = randInt(0, copy.length - 1);
    out.push(copy.splice(idx,1)[0]);
  }
  return out;
}

function renderMain(item, type){
  $("#rProvider").textContent = `${item.provider} (${item.code}) • Jam Ramai ${item.jam}`;
  $("#rSlotType").textContent = type;
  $("#rPercent").textContent = `${item.rtp}%`;

  $("#rRtpValue").textContent = getRtpText(item.rtp);
  $("#rJam").textContent = item.jam;
  $("#rPola").textContent = getRandomPattern();
  $("#rStatusText").textContent = getRandomStatus();
  $("#rTip").textContent = "COPYRIGHT BY SEO DEWANAGA";
}

function renderAlts(list, type){
  const wrap = $("#rtpAltList");
  wrap.innerHTML = "";

  list.forEach(x => {
    const el = document.createElement("div");
    el.className = "rtp-alt-item";
    el.innerHTML = `
      <div class="g">${x.provider} (${x.code})</div>
      <div class="p">Mode ${type} • RTP <strong>${x.rtp}%</strong></div>
      <div class="r">Spin Meter: <strong>${getSpinMeter()}</strong></div>
    `;
    wrap.appendChild(el);
  });
}

function spinButton(btn, on){
  if(on){
    btn.disabled = true;
    btn.dataset.text = btn.innerHTML;
    btn.innerHTML = '<span class="icon">🔄</span> PROCESSING...';
  }else{
    btn.disabled = false;
    btn.innerHTML = btn.dataset.text || '<span class="icon">🎰</span> GENERATE RTP';
  }
}

/* ===============================
   INIT

(function init(){
  const sel = $("#providerFilter");

  uniqueProviders().forEach(p => {
    const option = document.createElement("option");
    option.value = p;
    option.textContent = p;
    sel.appendChild(option);
  });

  $("#btnRtpRandom").addEventListener("click", (e) => {
    const btn = e.currentTarget;
    const type = $("#slotTypeFilter").value;
    const filtered = applyFilters();
    const st = $("#rtpStatus");

    if(!filtered.length){
      st.textContent = "— NO MATCH —";
      st.style.color = "#ffffff";
      renderAlts([], type);
      $("#rProvider").textContent = "-";
      $("#rSlotType").textContent = type;
      $("#rPercent").textContent = "-";
      $("#rRtpValue").textContent = "-";
      $("#rJam").textContent = "-";
      $("#rPola").textContent = "-";
      $("#rStatusText").textContent = "-";
      return;
    }

    spinButton(btn, true);
    st.textContent = "— SCANNING RTP —";
    st.style.animation = "pulse .6s linear infinite";
    st.style.color = "#ffffff";

    const t0 = setInterval(() => {
      const temp = pickRandom(filtered, 1)[0];
      $("#rProvider").textContent = `${temp.provider} (${temp.code}) • Jam Ramai ${temp.jam}`;
      $("#rSlotType").textContent = type;
      $("#rPercent").textContent = `${temp.rtp}%`;
      $("#rRtpValue").textContent = getRtpText(temp.rtp);
      $("#rJam").textContent = temp.jam;
      $("#rPola").textContent = getRandomPattern();
      $("#rStatusText").textContent = getRandomStatus();
    }, 85);

    setTimeout(() => {
      clearInterval(t0);

      const picks = pickRandom(filtered, 4);
      renderMain(picks[0], type);
      renderAlts(picks.slice(1), type);

      st.textContent = "— READY —";
      st.style.animation = "";
      st.style.color = "#ffc33b";
      spinButton(btn, false);
    }, 1200);
  });

  renderAlts([], "HIGH");
})();
